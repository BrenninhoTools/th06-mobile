function(th06_embed_bytes input name out_var)
    file(READ "${input}" hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1, " bytes "${hex}")
    set(${out_var} "\nstatic char ${name}[] = {${bytes}0};\n" PARENT_SCOPE)
endfunction()

th06_embed_bytes("${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert" vertShaderBytes vert_embed)
th06_embed_bytes("${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag" fragShaderBytes frag_embed)
file(WRITE "${TH06_EMBEDS}.tmp" "#pragma once\n${vert_embed}${frag_embed}")
configure_file("${TH06_EMBEDS}.tmp" "${TH06_EMBEDS}" COPYONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert"
    "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag")
